cmake_minimum_required(VERSION 3.18)

# Setup
if(ENABLE_CUDA)
    project(PyKokkosModule CXX CUDA)
elseif(ENABLE_HIP)
    project(PyKokkosModule CXX HIP)
else()
    project(PyKokkosModule CXX)
endif()

set(CMAKE_CXX_STANDARD ${CXX_STANDARD})
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

# Dependencies
find_package(Python3 COMPONENTS Interpreter Development REQUIRED)
find_package(pybind11 CONFIG REQUIRED)
find_package(OpenMP)

if(KOKKOS_LIB_PATH)
    get_filename_component(Kokkos_ROOT "${KOKKOS_LIB_PATH}" DIRECTORY)
endif()
find_package(Kokkos REQUIRED)
enable_language(${Kokkos_COMPILE_LANGUAGE})

# Sources and targets
file(GLOB SOURCE_FILES "*.cpp")
add_library(${MODULE_NAME} SHARED ${SOURCE_FILES})

set_target_properties(${MODULE_NAME} PROPERTIES
    PREFIX ""
    POSITION_INDEPENDENT_CODE ON
)

# Includes
target_include_directories(${MODULE_NAME} PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/..
)

# Compiler definitions and flags
target_compile_definitions(${MODULE_NAME} PRIVATE
    SPACE=${EXEC_SPACE}
    pk_arg_memspace=${PK_ARG_MEMSPACE}
    pk_arg_layout=${PK_ARG_LAYOUT}
    pk_exec_space=Kokkos::${EXEC_SPACE}
    pk_real=${PK_REAL}
)

if(ENABLE_HIP)
    target_compile_options(${MODULE_NAME} PRIVATE -fno-gpu-rdc)
endif()

# Link libraries
target_link_libraries(${MODULE_NAME} PRIVATE pybind11::module Python3::Python Kokkos::kokkos)
target_link_libraries(${MODULE_NAME} PRIVATE Kokkos::kokkoscore Kokkos::kokkoscontainers)

# Install
install(TARGETS ${MODULE_NAME} LIBRARY DESTINATION .)
